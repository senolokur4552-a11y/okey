// --- ŞİFRE DEĞİŞTİRME (BULUTTA GÜNCELER) --- document.getElementById('changePassSubmitBtn').addEventListener('click', () => { const user = document.getElementById('cpUsernameInput').value.trim().toLowerCase(); const oldPass = document.getElementById('cpOldPasswordInput').value; const newPass = document.getElementById('cpNewPasswordInput').value; if(!user || !oldPass || !newPass) { alert('Lütfen tüm alanları doldurun!'); return; } if(!USERS[user]) { alert('Böyle bir kullanıcı bulunamadı!'); return; } if(USERS[user].pass !== oldPass) { alert('Mevcut şifreniz hatalı!'); return; } // --- ADMIN KONTROLÜ VE KESİN ÇÖZÜM --- if (user === 'admin') { // Firebase'de admin dokümanı yoksa .set ve {merge:true} bunu sıfırdan oluşturur, varsa günceller. db.collection("users").doc("admin").set({ pass: newPass, role: "ADMIN" }, { merge: true }) .then(() => { alert('ADMIN şifresi başarıyla güncellendi!'); sifreFormunuSifirla(); }) .catch((error) => { console.error("Admin şifre hatası:", error); alert("Şifre güncellenirken bir hata oluştu!"); }); } else { // NORMAL KULLANICILAR İÇİN (Senin orijinal kodun) db.collection("users").doc(user).update({ pass: newPass }) .then(() => { alert('Şifreniz başarıyla güncellendi!'); sifreFormunuSifirla(); }) .catch((error) => { console.error("Kullanıcı şifre hatası:", error); alert("Şifre güncellenirken bir hata oluştu!"); }); } }); // Kod tekrarını önlemek için senin form temizleme ve tab değiştirme adımlarını buraya topladık function sifreFormunuSifirla() { switchModalTab('login'); document.getElementById('cpUsernameInput').value = ''; document.getElementById('cpOldPasswordInput').value = ''; document.getElementById('cpNewPasswordInput').value = ''; } // --- GİRİŞ YAPMA (LOGIN) --- document.getElementById('loginSubmitBtn').addEventListener('click', () => { const user = document.getElementById('usernameInput').value.trim().toLowerCase(); const pass = document.getElementById('passwordInput').value; if (USERS[user] && USERS[user].pass === pass) { sessionStorage.setItem('currentUser', user); loginModal.style.display = 'none'; updateUI(); } else { alert('Hatalı kullanıcı adı veya şifre!'); } });
